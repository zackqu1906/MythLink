022  def split_response(raw):
023      """Accept body on the header line or next line, preserving body contents."""
024      match = re.fullmatch(r'分类：([^\r\n]+)\r?\n正文：([\s\S]*)', raw)
025      if not match:
026          raise ValueError('Missing classification line or body header')
027      body = match.group(2)
028      if body.startswith('\r\n'):
029          body = body[2:]
030      elif body.startswith('\n'):
031          body = body[1:]
032      return match.group(1), body
