cmake_minimum_required(VERSION 3.20)
project(MythlinkTsf LANGUAGES CXX RC)

add_library(MythlinkTsf SHARED TextService.cpp MythlinkTsf.def MythlinkTsf.rc)
target_include_directories(MythlinkTsf PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(MythlinkTsf PRIVATE cxx_std_17)
target_link_libraries(MythlinkTsf PRIVATE ole32 oleaut32 advapi32 uuid)
set_target_properties(MythlinkTsf PROPERTIES PREFIX "" MSVC_RUNTIME_LIBRARY "MultiThreaded")
if(MSVC)
    target_compile_options(MythlinkTsf PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/utf-8>)
elseif(MINGW)
    target_link_options(MythlinkTsf PRIVATE -static)
endif()
